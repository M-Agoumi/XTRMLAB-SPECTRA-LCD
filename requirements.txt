# hongtai_screen -- everything needed to run the desktop app (app.py)
# with all four themes available. Windows only.
#
# Only want one theme instead of the full desktop app? You don't need
# all of this -- see that theme's own section in README.md for just
# its dependencies.
#
#   pip install -r requirements.txt

# --- Core: the driver itself + the desktop app ------------------------
pyserial>=3.5
pillow>=10.0          # ImageOps is used for the Dashboard's custom-image background

# --- demo_clock.py: CPU/RAM bars ---------------------------------------
psutil>=5.9

# --- dashboard_theme.py: gauges, CPU/GPU stats, Spotify art -----------
numpy>=1.24          # required -- gauge rendering + background gradients
pycairo>=1.24        # required -- gauge rendering (prebuilt wheel on Windows)
nvidia-ml-py>=12.0   # optional -- NVIDIA GPU/VRAM/power stats; degrades gracefully without one
winsdk>=1.0.4        # optional -- Spotify/now-playing album art (Windows 10 1809+)

# --- video_theme.py: stream a video file to the panel ------------------
opencv-python-headless>=4.8
# pygame            # optional -- only needed for --audio; also needs ffmpeg on PATH

# --- webpage_theme.py: mirror a webpage to the panel --------------------
playwright>=1.40
# after installing, run once:  playwright install chromium   (~150MB download)

# --- Desktop app extra: system tray icon on close/at startup -----------
pystray>=0.19        # optional -- without it, closing the window quits instead
                     # of dismissing to tray, and --autostart just minimizes

# --- Building a standalone .exe (developers only, not needed to run) --
# pyinstaller>=6.0  # see BUILD.md -- uncomment or `pip install pyinstaller` yourself
